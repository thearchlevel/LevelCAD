cmake_minimum_required(VERSION 3.21)
project(LevelCAD VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(LEVELCAD_BUILD_DESKTOP "Build Qt desktop application" ON)
option(LEVELCAD_BUILD_TESTS "Build unit tests" ON)
add_subdirectory(engine)
if(LEVELCAD_BUILD_DESKTOP)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Concurrent)
  add_subdirectory(application/desktop)
endif()
if(LEVELCAD_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
